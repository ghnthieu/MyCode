     sn=integer;

begin

  for i:=1 to n do
    for j:=1 to m do
      begin
        write('A[',i,',',j,']= '); readln(A[i,j]);

procedure in_ra(a:mang; n,m:sn);
begin
  for i:=1 to n do
   begin
    for j:=1 to m do
      write(a[i,j],' ');
    writeln;
   end;
  writeln('---------------');
end;
