program tong_n;
uses crt;
var n:integer;
procedure nhap(var n:integer);
begin
  write('nhap 1 so: ');readln(n);
end;
procedure xuat_tong_so;
var i,tong:integer;
begin
  i:=1;
  while i<n do
    begin
      tong:=tong+i;
      inc(i);
    end;
writeln('tong tu 1 den n la: ',tong);
end;

BEGIN
  clrscr;
  nhap(n);
  xuat_tong_so;
readln
END.
