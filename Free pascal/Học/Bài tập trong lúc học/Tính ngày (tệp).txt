program bt;
uses crt;
var thang,nam:integer;
    fi,fo:text;
begin
  clrscr;
  assign(fi,'songay.inp');reset(fi);
  read(fi,thang,nam);
  assign(fo,'songay.out');rewrite(fo);
  write('nhap 1 thang: ');readln(thang);
  case thang of
    1,3,5,7,8,10,12:writeln(fo,'thang ',thang,' co 31 ngay');
    2:begin
        write('nhap 1 nam: ');readln(nam);
        if (nam mod 4=0) and (nam mod 100<>0) or (nam mod 400=0) then
          writeln(fo,'la nam nhuan co 29 ngay')
        else writeln(fo,'la nam thuong co 28 ngay');
      end;
    4,6,9,11:writeln(fo,'thang ',thang,' co 30 ngay');
    else writeln(fo,'day khong phai la 1 thang');
  end;
  close(fi);
  close(fo);
end.
