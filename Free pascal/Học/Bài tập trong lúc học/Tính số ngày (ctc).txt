program so_ngay;
uses crt;
var nam,thang:integer;
procedure nhap(var n:integer);
begin
  write('nhap 1 thang: ');readln(thang);
end;
procedure xuat_so_ngay;
begin
  case thang of
    1,3,5,7,8,10,12:writeln('thang ',thang,' co 31 ngay');
    2:begin
        write('nhap 1 nam: ');readln(nam);
        if (nam mod 4=0) and (nam mod 100<>0) or (nam mod 400=0) then
          writeln('la nam nhuan co 29 ngay')
        else writeln('la nam thuong co 28 ngay');
      end;
    4,6,9,11:writeln('thang ',thang,' co 30 ngay');
    else writeln('day khong phai la 1 thang');
  end;
end;

BEGIN
  clrscr;
  nhap(thang);
  xuat_so_ngay;
readln
END.
