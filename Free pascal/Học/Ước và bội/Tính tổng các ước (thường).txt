program tong_uoc;
uses crt;
var dem,dem_i,n,i:word;
    tonguoc:qword;
begin
clrscr;
  write('nhap 1 so: ');readln(n);
  write('uoc cua ',n,' la: ');
  tonguoc:=0;
  dem:=0;
  dem_i:=0;
  for i:=1 to n div 2 do
    if n mod i=0 then
      inc(dem);
  for i:=1 to n div 2 do
    if n mod i=0 then
      begin
        inc(dem_i);
        write(i);
        if dem_i<dem then
          write('+')
        else
          write('=');
        tonguoc:=tonguoc+i;
      end;
  write(tonguoc);
readln
end.
