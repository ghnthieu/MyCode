Program so_nt;
uses crt;
var n,i,j,k,ka,kb:byte;
    a,b:array[1..100] of longint;
function kt(n:longint):boolean;
var ok:boolean;
    i:longint;
begin
  ok:=true;
  if n<2 then
    ok:=false
  else
    for i:=2 to trunc(sqrt(n)) do
      if n mod i=0 then
        begin
          ok:=false;
          break;
        end;
  kt:=ok;
end;
Begin
  clrscr;
  repeat
    write('nhap do dai cua so: ');readln(n);
  until (n>0) and (n<10);
  ka:=1;
  a[ka]:=0;
  for i:=1 to n do
    begin
      kb:=0;
      for k:=1 to ka do
        for j:=0 to 9 do
          if kt(a[k]*10+j) then
            begin
              inc(kb);
              b[kb]:=a[k]*10+j;
            end;
      ka:=kb;
      for k:=1 to ka do
        a[k]:=b[k];
    end;
  for k:=1 to ka do
    write(a[k],' ');
  readln
End.
